cmake_minimum_required(VERSION 3.25)
project(kdu_embedded_probe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This isolated probe requires 64-bit Windows")
endif()
add_executable(kdu_runtime_probe runtime_probe.cpp)
target_compile_features(kdu_runtime_probe PRIVATE cxx_std_20)
target_compile_definitions(kdu_runtime_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)

add_executable(kdu_decode_probe decode_probe.cpp)
target_compile_features(kdu_decode_probe PRIVATE cxx_std_20)
target_compile_definitions(kdu_decode_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(kdu_decode_probe PRIVATE bcrypt)

set(NEOJ2K_CPP_SOURCE "" CACHE PATH "neo-j2k C++ source checkout for matched comparison")
if(NEOJ2K_CPP_SOURCE)
  set(NJ2K_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(NJ2K_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(NJ2K_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(NJ2K_BUILD_BENCH OFF CACHE BOOL "" FORCE)
  set(NJ2K_INSTALL OFF CACHE BOOL "" FORCE)
  set(NJ2K_SIMD AVX2 CACHE STRING "" FORCE)
  add_subdirectory("${NEOJ2K_CPP_SOURCE}" cpp)
  add_executable(kdu_compare_native compare_native.cpp)
  target_compile_features(kdu_compare_native PRIVATE cxx_std_20)
  target_compile_definitions(kdu_compare_native PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(kdu_compare_native PRIVATE /EHsc /arch:AVX2)
  target_link_libraries(kdu_compare_native PRIVATE bcrypt neoj2k::static)
endif()
