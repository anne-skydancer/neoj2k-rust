cmake_minimum_required(VERSION 3.25)
project(nj2k_audit_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NJ2K_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(NJ2K_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(NJ2K_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NJ2K_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NJ2K_BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(NJ2K_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("C:/Users/annes/source/repos/neo-j2k" codec)
add_library(nj2k_audit_bridge STATIC audit_bridge.cpp)
target_include_directories(nj2k_audit_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(nj2k_audit_bridge PRIVATE "C:/Users/annes/source/repos/neo-j2k/src")
target_link_libraries(nj2k_audit_bridge PUBLIC neoj2k_static)
if(MSVC)
 target_compile_options(nj2k_audit_bridge PRIVATE /fp:precise /EHsc /Ob2)
 if(NJ2K_SIMD STREQUAL "AVX2")
  target_compile_options(nj2k_audit_bridge PRIVATE /arch:AVX2)
 endif()
else()
 target_compile_options(nj2k_audit_bridge PRIVATE -ffp-contract=off -fno-fast-math)
endif()
nj2k_no_ipo(nj2k_audit_bridge)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/audit-libs-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:nj2k_audit_bridge>\n$<TARGET_FILE:neoj2k_static>\n")
