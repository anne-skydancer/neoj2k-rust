use core::hint::select_unpredictable;

use super::tables::MQ_STATES;

pub(super) const CONTEXT_AGGREGATE: usize = 17;
pub(super) const CONTEXT_UNIFORM: usize = 18;
const MPS_BIT: u32 = 16;
const NEXT_MPS_SHIFT: u32 = 17;
const NEXT_LPS_SHIFT: u32 = 24;
const STATE_COUNT: usize = 128;
// A context index is masked to this many slots (the 19 contexts and padding), so that no access is out of bounds.
const CONTEXT_SLOTS: usize = 32;

// Cache the active state in each context. A symbol that leaves its context unchanged
// can use Qe and MPS immediately, with no separate state-table lookup.
static PACKED_STATES: [u32; STATE_COUNT] = {
    let mut table = [0; STATE_COUNT];
    let mut i = 0;
    while i < MQ_STATES.len() {
        let (qeval, mps, next_mps, next_lps) = MQ_STATES[i];
        assert!(qeval <= 0xffff && mps <= 1);
        assert!((next_mps as usize) < MQ_STATES.len() && (next_lps as usize) < MQ_STATES.len());
        table[i] =
            qeval | (mps << MPS_BIT) | ((next_mps as u32) << NEXT_MPS_SHIFT) | ((next_lps as u32) << NEXT_LPS_SHIFT);
        i += 1;
    }
    table
};

const INITIAL_CONTEXTS: [u32; CONTEXT_SLOTS] = {
    let mut contexts = [PACKED_STATES[0]; CONTEXT_SLOTS];
    contexts[0] = PACKED_STATES[8];
    contexts[CONTEXT_AGGREGATE] = PACKED_STATES[6];
    contexts[CONTEXT_UNIFORM] = PACKED_STATES[92];
    contexts
};

/// The byte source of a codeword segment: the bytes not read yet and the last one read. Only the byte-in path, which
/// a decision reaches 0.07 to 0.11 of the time, and the raw reader touch it.
#[derive(Clone, Copy, Debug)]
struct Source<'a> {
    data: &'a [u8],
    current: u32,
}

impl Source<'_> {
    /// BYTEIN of ISO 15444-1 C.3.4: the next byte into the code register `c` (a 0xff before a byte above 0x8f is a
    /// marker and gives 0xff00 with no byte taken). Returns the code register and the count of bits it now holds.
    fn byte_in(&mut self, c: u32) -> (u32, u32) {
        let (next, rest) = next_byte(self.data);
        if self.current != 0xff {
            self.data = rest;
            self.current = next;
            (c.wrapping_add(next << 8), 8)
        } else if next > 0x8f {
            (c.wrapping_add(0xff00), 8)
        } else {
            self.data = rest;
            self.current = next;
            (c.wrapping_add(next << 9), 7)
        }
    }
}

fn next_byte(data: &[u8]) -> (u32, &[u8]) {
    // The two artificial 0xff bytes form a marker. The existing byte-in logic stops on it.
    data.split_first()
        .map_or((0xff, data), |(&byte, rest)| (u32::from(byte), rest))
}

/// The three registers of the decision: the interval `a`, the code register `c` and the count `bits` of bits `c` can
/// still shift before a byte is needed. They are the only state that changes at every decision.
#[derive(Clone, Copy, Debug)]
struct Registers {
    a: u32,
    c: u32,
    bits: u32,
}

/// RENORMD when the code register has fewer bits than the interval needs shifts (a byte comes in: 0.07 to 0.11 of a
/// decision): the shifts taken as many at a time as the register has bits, a byte read whenever none is left. The
/// helper is outlined and cold. The two registers it changes go in as plain integers and come back as one `u64` (the
/// code register in the low half, the bit count in the high half), so that no three-word aggregate is built, passed by
/// address and copied back around the call; the interval `a` is not passed at all, the caller shifts it by `total`.
/// That is the intent: whether the compiler keeps these values in machine registers depends on the target and is not
/// relied on for correctness.
#[cold]
#[inline(never)]
fn renormalize_with_bytes(mut c: u32, mut bits: u32, source: &mut Source<'_>, total: u32) -> u64 {
    let mut shifts = total;
    while shifts > bits {
        c <<= bits;
        shifts -= bits;
        (c, bits) = source.byte_in(c);
    }
    c <<= shifts;
    bits -= shifts;
    u64::from(c) | (u64::from(bits) << 32)
}

/// The code register and the bit count that `renormalize_with_bytes` packed into one `u64`.
#[inline(always)]
#[allow(
    clippy::cast_possible_truncation,
    reason = "the two halves are taken apart on purpose: the low 32 bits are the code register, the high 32 the bit count, both built from u32 values"
)]
const fn unpack_registers(packed: u64) -> (u32, u32) {
    (packed as u32, (packed >> 32) as u32)
}

/// One code-block's arithmetic state, reused across its codeword segments: the registers, the byte source and the
/// contexts, in three fields that a pass takes apart (`enter`), so that the registers are plain locals of the pass.
#[derive(Clone, Copy, Debug)]
pub(super) struct Decoder<'a> {
    source: Source<'a>,
    registers: Registers,
    contexts: [u32; CONTEXT_SLOTS],
}

/// A pass's view of the decoder: the registers as locals (held in machine registers for the whole pass: they are
/// never indexed or addressed, and the context array is a separate object they do not share a block with), and the
/// byte source and the contexts by reference. `leave` writes the registers back.
#[derive(Debug)]
pub(super) struct Mq<'d, 'a> {
    a: u32,
    c: u32,
    bits: u32,
    source: &'d mut Source<'a>,
    contexts: &'d mut [u32; CONTEXT_SLOTS],
    home: &'d mut Registers,
}

impl<'a> Decoder<'a> {
    pub(super) fn new() -> Self {
        Self {
            source: Source { data: &[], current: 0 },
            registers: Registers {
                a: 0x8000,
                c: 0,
                bits: 0,
            },
            contexts: INITIAL_CONTEXTS,
        }
    }

    pub(super) fn reset_contexts(&mut self) {
        self.contexts = INITIAL_CONTEXTS;
    }

    pub(super) fn start_segment(&mut self, data: &'a [u8], raw: bool) {
        self.source.data = data;
        let r = &mut self.registers;
        if raw {
            r.c = 0;
            r.bits = 0;
        } else {
            (self.source.current, self.source.data) = next_byte(data);
            r.c = self.source.current << 16;
            (r.c, r.bits) = self.source.byte_in(r.c);
            r.c <<= 7;
            r.bits -= 7;
            r.a = 0x8000;
        }
    }

    /// Takes the registers into locals for a pass; the pass ends with `Mq::leave`.
    #[inline(always)]
    pub(super) fn enter(&mut self) -> Mq<'_, 'a> {
        let Self {
            source,
            registers,
            contexts,
        } = self;
        Mq {
            a: registers.a,
            c: registers.c,
            bits: registers.bits,
            source,
            contexts,
            home: registers,
        }
    }

    #[cfg(test)]
    fn symbol(&mut self, context: usize) -> u32 {
        let mut mq = self.enter();
        let bit = mq.symbol(context);
        mq.leave();
        bit
    }

    #[cfg(test)]
    fn aggregate_run(&mut self, flags: &[Cell<u32>]) -> usize {
        let mut mq = self.enter();
        let run = mq.aggregate_run(flags);
        mq.leave();
        run
    }
}

impl Mq<'_, '_> {
    #[inline(always)]
    pub(super) fn leave(self) {
        *self.home = Registers {
            a: self.a,
            c: self.c,
            bits: self.bits,
        };
    }

    /// The decision of ISO 15444-1 C.3.2 with its renormalisation, as the C++'s `opj_mqc_decode_macro`: the branches
    /// the decoded bit does not steer are selects, the renormalisation is always run (no shift when bit 15 of the
    /// interval is set) and only a byte coming in branches.
    ///
    /// `lps` says the code register is in the LPS sub-interval; `flip` that the symbol is the less probable one (an
    /// LPS-interval hit the conditional exchange leaves an MPS, or an MPS-interval hit it makes an LPS). The state
    /// changes, to the next-LPS state on a flip and else to the next-MPS state, when the interval is the LPS one or has
    /// lost bit 15.
    #[inline(always)]
    pub(super) fn symbol(&mut self, context: usize) -> u32 {
        let slot = context & (CONTEXT_SLOTS - 1);
        let word = self.contexts[slot];
        let qeval = word & 0xffff;
        let a = self.a - qeval;
        let lps = self.c >> 16 < qeval;
        let flip = lps ^ (a < qeval);
        let bit = ((word >> MPS_BIT) & 1) ^ u32::from(flip);
        // Selects the compiler is told are unpredictable (it otherwise turns them back into branches).
        let next_shift = select_unpredictable(flip, NEXT_LPS_SHIFT, NEXT_MPS_SHIFT);
        let next = PACKED_STATES[((word >> next_shift) as usize) & (STATE_COUNT - 1)];
        let changes = lps | (a < 0x8000);
        self.contexts[slot] = select_unpredictable(changes, next, word);
        self.c -= select_unpredictable(lps, 0, qeval << 16);
        let a = select_unpredictable(lps, qeval, a);
        // the renormalisation: the shifts at once when the code register has the bits
        let total = a.leading_zeros() - 16;
        if total <= self.bits {
            self.a = a << total;
            self.c <<= total;
            self.bits -= total;
        } else {
            let (c, bits) = unpack_registers(renormalize_with_bytes(self.c, self.bits, self.source, total));
            self.a = a << total;
            self.c = c;
            self.bits = bits;
        }
        bit
    }

    /// The columns of the cleanup pass's run of empty-stripe columns that the AGG context decodes as 0 without a
    /// renormalisation, taken at once: the C++'s `opj_mqc_agg_run_macro`. `flags` is the flag words from the current
    /// column to the end of its row. A column of an empty stripe asks the AGG context first; when its MPS is 0 and
    /// the interval and the code register have room for it (a decision that is an MPS and keeps bit 15 of `a`:
    /// `a - qe >= 0x8000` and `c >> 16 >= qe`), it decodes 0 and the pass goes on to the next column. Step j of such a
    /// run needs `j * qe <= a - 0x8000` and `j * qe <= c >> 16`: the run lasts while `n * qe <= room`. The `n` columns
    /// (0: no run, the caller decides as usual) take `a -= n * qe` and `c -= (n * qe) << 16` at once: no state
    /// change, no byte.
    #[inline(always)]
    pub(super) fn aggregate_run(&mut self, flags: &[Cell<u32>]) -> usize {
        let word = self.contexts[CONTEXT_AGGREGATE & (CONTEXT_SLOTS - 1)];
        let qeval = word & 0xffff;
        let room = (self.a - 0x8000).min(self.c >> 16);
        if (word >> MPS_BIT) & 1 != 0 || qeval > room {
            return 0;
        }
        let mut used = qeval;
        let mut n = 1;
        while n < flags.len() && flags[n].get() == 0 && used + qeval <= room {
            used += qeval;
            n += 1;
        }
        self.a -= used;
        self.c -= used << 16;
        n
    }

    #[inline(always)]
    pub(super) fn raw(&mut self) -> u32 {
        if self.bits == 0 {
            let (next, rest) = next_byte(self.source.data);
            if self.c == 0xff && next > 0x8f {
                self.bits = 8;
            } else {
                self.bits = if self.c == 0xff { 7 } else { 8 };
                self.c = next;
                self.source.data = rest;
            }
        }
        self.bits -= 1;
        (self.c >> self.bits) & 1
    }
}
