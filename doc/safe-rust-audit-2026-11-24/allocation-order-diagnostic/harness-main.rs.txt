use std::time::{Instant,Duration};
macro_rules! module{($m:ident,$k:ident)=>{mod $m{const SLOT:&str=stringify!($m);use $k as neo_j2k; include!("suite.rs");}};}
module!(base,neo_j2k_old);module!(candidate,neo_j2k_new);module!(control,neo_j2k_control);
fn median(mut x:Vec<f64>)->f64{x.sort_by(f64::total_cmp);x[x.len()/2]}
fn timings(mut calls:[&mut dyn FnMut();3],batch:u32)->[f64;5]{
 let mut times:[Vec<f64>;3]=std::array::from_fn(|_|Vec::new());let mut gains=Vec::new();let mut controls=Vec::new();
 let orders=[[0,1,2],[2,1,0],[1,0,2],[2,0,1],[0,2,1],[1,2,0]];
 for n in 0..24{let mut t=[0.;3];for k in orders[n%6]{let begin=Instant::now();for _ in 0..batch{calls[k]();}t[k]=begin.elapsed().as_secs_f64()*1e6/f64::from(batch);times[k].push(t[k]);}gains.push((t[1]/t[0]-1.)*100.);controls.push((t[2]/t[0]-1.)*100.);}
 [median(std::mem::take(&mut times[0])),median(std::mem::take(&mut times[1])),median(std::mem::take(&mut times[2])),median(gains),median(controls)]
}
static ALLOCATION_ORDER: std::sync::atomic::AtomicUsize=std::sync::atomic::AtomicUsize::new(0);
fn decode(name:&str,reduce:u32,cut:bool,width:usize){
 let mut bytes=std::fs::read(format!("{}/../corpus/generated/{name}",env!("CARGO_MANIFEST_DIR"))).unwrap();if cut{bytes.truncate(bytes.len()-2);}
 let expected_cpp=if name.starts_with('p') {let mut codec=neo_j2k_oracle::Codec::new(neo_j2k_oracle::Format::J2k,reduce,0).unwrap();let(status,samples)=codec.decode_i32(&bytes);assert_eq!(status,0);Some(samples)}else{None};
 let pool=neo_j2k_oracle::BenchmarkPool::new(width.try_into().unwrap()).unwrap();
 let mut run=|a:&mut dyn FnMut(),ao:Vec<i32>,b:&mut dyn FnMut(),bo:Vec<i32>,c:&mut dyn FnMut(),co:Vec<i32>|{
  assert_eq!(ao,bo);assert_eq!(ao,co);if let Some(cpp)=&expected_cpp{assert_eq!(&ao,cpp);}
  let t=timings([a,b,c],if width==1&&!name.contains("large"){32}else{2});println!("decode|{name}|r{reduce}|cut{cut}|w{width}|{:.4}|{:.4}|{:.4}|{:+.3}%|{:+.3}%|same",t[0],t[1],t[2],t[3],t[4]);
 };
 match ALLOCATION_ORDER.load(std::sync::atomic::Ordering::Relaxed){
0=>{base::decode(&bytes,reduce,&pool,|a,ao|{candidate::decode(&bytes,reduce,&pool,|b,bo|{control::decode(&bytes,reduce,&pool,|c,co|{run(a,ao,b,bo,c,co)})})})},
1=>{control::decode(&bytes,reduce,&pool,|c,co|{candidate::decode(&bytes,reduce,&pool,|b,bo|{base::decode(&bytes,reduce,&pool,|a,ao|{run(a,ao,b,bo,c,co)})})})},
2=>{candidate::decode(&bytes,reduce,&pool,|b,bo|{base::decode(&bytes,reduce,&pool,|a,ao|{control::decode(&bytes,reduce,&pool,|c,co|{run(a,ao,b,bo,c,co)})})})},
3=>{control::decode(&bytes,reduce,&pool,|c,co|{base::decode(&bytes,reduce,&pool,|a,ao|{candidate::decode(&bytes,reduce,&pool,|b,bo|{run(a,ao,b,bo,c,co)})})})},
4=>{base::decode(&bytes,reduce,&pool,|a,ao|{control::decode(&bytes,reduce,&pool,|c,co|{candidate::decode(&bytes,reduce,&pool,|b,bo|{run(a,ao,b,bo,c,co)})})})},
5=>{candidate::decode(&bytes,reduce,&pool,|b,bo|{control::decode(&bytes,reduce,&pool,|c,co|{base::decode(&bytes,reduce,&pool,|a,ao|{run(a,ao,b,bo,c,co)})})})},_=>unreachable!()}
}

fn packed(name:&str,reduce:u32,channels:usize){
 let bytes=std::fs::read(format!("{}/../corpus/generated/{name}",env!("CARGO_MANIFEST_DIR"))).unwrap();
 base::packed(&bytes,reduce,channels,|a,ao|candidate::packed(&bytes,reduce,channels,|b,bo|control::packed(&bytes,reduce,channels,|c,co|{assert_eq!(ao,bo);assert_eq!(ao,co);let t=timings([a,b,c],16);println!("packed|{name}|r{reduce}|channels{channels}|{:.4}|{:.4}|{:.4}|{:+.3}%|{:+.3}%|same",t[0],t[1],t[2],t[3],t[4]);})));
}

fn encode_one(side:usize,reversible:bool){let pixels=base::pixels(side);
 base::encode(&pixels,side as u32,reversible,|a,ao|candidate::encode(&pixels,side as u32,reversible,|b,bo|control::encode(&pixels,side as u32,reversible,|c,co|{
 assert_eq!(ao,bo);assert_eq!(ao,co);let t=timings([a,b,c],32);println!("encode|{side}|reversible{reversible}|{:.4}|{:.4}|{:.4}|{:+.3}%|{:+.3}%|same",t[0],t[1],t[2],t[3],t[4]);})));
}
fn main(){
 println!("ALLOCATION ORDER DIAGNOSTIC: one executable, original/candidate/unchanged-control; only construction/warm order varies at runtime. Timed order remains balanced.");
 for pass in 0..2 {let orders=if pass==0 {[0,1,2,3,4,5]}else{[5,3,1,4,2,0]};for order in orders {
  ALLOCATION_ORDER.store(order,std::sync::atomic::Ordering::Relaxed);
  println!("START|pass{pass}|order{order}");
  decode("p31_large_1024_rgb_97.j2k",0,false,1);
  decode("p24_sl_128_rgba_97.j2k",0,false,1);
  decode("p31_sl_128_rgba_97.j2k",0,false,1);
  decode("sl_64_rgb_97.j2k",0,false,1);
  encode_one(32,true);encode_one(32,false);
  println!("DONE|pass{pass}|order{order}|6rows");
 }}
}
